class AutoClicker:
    def __init__(self):
        self.running = False
        self.activation_key = None
        self.cps = 10.0
        self.duty = 100.0
        self.hold_mode = True
        self.capturing_key = False

        # NEW: left/right click selector
        self.click_side = "left"

        self.load_settings()

        self.key_listener = KeyListener(on_press=self.on_key_press, on_release=self.on_key_release)
        self.key_listener.start()

        from pynput.mouse import Listener as MouseListener
        self.mouse_listener = MouseListener(on_click=self.on_mouse_click)
        self.mouse_listener.start()

        threading.Thread(target=self.click_loop, daemon=True).start()

    # --------------------------
    # SETTINGS SAVE / LOAD
    # --------------------------
    def save_settings(self):
        data = {
            "activation_key": self.activation_key,
            "cps": self.cps,
            "duty": self.duty,
            "hold_mode": self.hold_mode,
            "click_side": self.click_side
        }
        with open(SETTINGS_FILE, "w") as f:
            json.dump(data, f)

    def load_settings(self):
        if not os.path.exists(SETTINGS_FILE):
            self.activation_key = None
            self.cps = 10.0
            self.duty = 100.0
            self.hold_mode = True
            self.click_side = "left"
            return

        try:
            with open(SETTINGS_FILE, "r") as f:
                data = json.load(f)
                self.activation_key = data.get("activation_key", None)
                self.cps = float(data.get("cps", 10.0))
                self.duty = float(data.get("duty", 100.0))
                self.hold_mode = bool(data.get("hold_mode", True))
                self.click_side = data.get("click_side", "left")
        except:
            self.activation_key = None
            self.cps = 10.0
            self.duty = 100.0
            self.hold_mode = True
            self.click_side = "left"

    # --------------------------
    # Activation Key Capture
    # --------------------------
    def start_capture(self):
        self.capturing_key = True

    def finish_capture(self, key_name):
        self.activation_key = key_name
        self.capturing_key = False
        self.save_settings()

    # --------------------------
    # Keyboard Listener
    # --------------------------
    def on_key_press(self, key):
        if self.capturing_key:
            try:
                if isinstance(key, KeyCode):
                    self.finish_capture(key.char)
                else:
                    self.finish_capture(str(key))
            except:
                self.finish_capture(str(key))
            return

        if self.activation_key is None:
            return

        if self.hold_mode:
            if self.key_matches(key):
                self.running = True
        else:
            if self.key_matches(key):
                self.running = not self.running

    def on_key_release(self, key):
        if self.hold_mode and self.key_matches(key):
            self.running = False

    # --------------------------
    # Mouse Listener
    # --------------------------
    def on_mouse_click(self, x, y, button, pressed):
        mapping = {
            Button.left: "mouse_left",
            Button.right: "mouse_right",
            Button.middle: "mouse_middle",
            Button.x1: "mouse_side_forward",
            Button.x2: "mouse_side_back"
        }

        if self.capturing_key:
            self.finish_capture(mapping.get(button, str(button)))
            return

        if self.activation_key is None:
            return

        if self.hold_mode:
            if self.mouse_matches(button):
                self.running = pressed
        else:
            if self.mouse_matches(button) and pressed:
                self.running = not self.running

    # --------------------------
    # Matching Logic
    # --------------------------
    def key_matches(self, key):
        try:
            if isinstance(key, KeyCode):
                return key.char == self.activation_key
        except:
            pass
        return str(key) == self.activation_key

    def mouse_matches(self, button):
        mapping = {
            Button.left: "mouse_left",
            Button.right: "mouse_right",
            Button.middle: "mouse_middle",
            Button.x1: "mouse_side_forward",
            Button.x2: "mouse_side_back"
        }
        return mapping.get(button) == self.activation_key

    # --------------------------
    # CLICK LOOP
    # --------------------------
    def click_loop(self):
        while True:
            if self.running:
                base_cps = max(0.01, min(self.cps, 500))

                if self.duty <= 0:
                    effective_cps = base_cps
                else:
                    effective_cps = base_cps * (self.duty / 100.0)

                effective_cps = max(0.01, effective_cps)
                period = 1.0 / effective_cps

                next_click = time.perf_counter()

                while self.running:
                    if self.duty <= 0:
                        effective_cps = base_cps
                    else:
                        effective_cps = base_cps * (self.duty / 100.0)

                    effective_cps = max(0.01, effective_cps)
                    period = 1.0 / effective_cps

                    now = time.perf_counter()
                    if now >= next_click:

                        if self.click_side == "left":
                            mouse.press(Button.left)
                            mouse.release(Button.left)
                        else:
                            mouse.press(Button.right)
                            mouse.release(Button.right)

                        next_click += period
                    else:
                        time.sleep(0.0005)
            else:
                time.sleep(0.01)

    # --------------------------
    # Input Validation
    # --------------------------
    def validate_numeric_input(self, new_value):
        if new_value == "":
            return True
        if new_value.count(".") > 1:
            return False
        for ch in new_value:
            if not (ch.isdigit() or ch == "."):
                return False
        return True

    # Discord Logo Loader
    # --------------------------
    def load_discord_logo(self):
        try:
            response = requests.get(DISCORD_LOGO_URL)
            img = Image.open(BytesIO(response.content))
            return ctk.CTkImage(light_image=img, dark_image=img, size=(28, 28))
        except:
            return None

    # --------------------------
    # UI Logic
    # --------------------------
    def start_capture(self):
        self.ac.start_capture()
        self.key_btn.configure(text="Press any key or mouse button...")

        def wait_update():
            while self.ac.capturing_key:
                time.sleep(0.05)
            if self.ac.activation_key:
                self.key_btn.configure(text=f"Activation: {self.ac.activation_key}")
            else:
                self.key_btn.configure(text="Set Activation Key")

        threading.Thread(target=wait_update, daemon=True).start()

    def change_mode(self, mode):
        self.ac.hold_mode = (mode == "Hold")
        self.ac.save_settings()

    def apply(self):
        try:
            cps = float(self.cps_entry.get())
        except:
            cps = 10.0

        try:
            duty = float(self.duty_entry.get())
        except:
            duty = 100.0

        if cps > 500:
            cps = 500
            self.cps_entry.delete(0, "end")
            self.cps_entry.insert(0, "500")

        if duty > 500:
            duty = 500
            self.duty_entry.delete(0, "end")
            self.duty_entry.insert(0, "500")

        cps = max(0.01, cps)
        duty = max(0.0, duty)

        self.ac.cps = cps
        self.ac.duty = duty
        self.ac.save_settings()

    # --------------------------
    # EXPORT / IMPORT
    # --------------------------
    def export_settings(self):
        file_path = filedialog.asksaveasfilename(
            defaultextension=".json",
            filetypes=[("JSON Files", "*.json")]
        )
        if not file_path:
            return

        data = {
            "activation_key": self.ac.activation_key,
            "cps": self.ac.cps,
            "duty": self.ac.duty,
            "hold_mode": self.ac.hold_mode,
            "click_side": self.ac.click_side
        }

        with open(file_path, "w") as f:
            json.dump(data, f)

    def import_settings(self):
        file_path = filedialog.askopenfilename(
            filetypes=[("JSON Files", "*.json")]
        )
        if not file_path:
            return

        try:
            with open(file_path, "r") as f:
                data = json.load(f)

            self.ac.activation_key = data.get("activation_key", None)
            self.ac.cps = float(data.get("cps", 10.0))

                text=f"Activation: {self.ac.activation_key}"
                if self.ac.activation_key else "Set Activation Key"

            self.cps_entry.insert(0, str(self.ac.cps))

            self.duty_entry.delete(0, "end")
            self.duty_entry.insert(0, str(self.ac.duty))

                pass
